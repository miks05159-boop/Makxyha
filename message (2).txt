-- воркер: тянет модули и данные, НИЧЕГО не создаёт через Instance
local function fetchPetsAsync(onDone)
	task.spawn(function()
		pcall(function()
			if not TGSPetSystem then
				TGSPetSystem = require(workspace.Lib.PetSystem.TGSPetSystem)
			end
			if next(BannedPets) == nil then
				local bp = require(workspace.Lib.Trading.TradeBannedPets)
				if type(bp) == "table" then BannedPets = bp end
			end
			local owned = TGSPetSystem.GetOwnedPets(plr) or {}
			local list = {}
			for id, pet in pairs(owned) do
				table.insert(list, {
					id = id,
					name = pet.Name,
					rarity = pet.Rarity,
					op = BannedPets[pet.Name] == true,
				})
			end
			table.sort(list, function(a, b)
				if a.op ~= b.op then return a.op end
				if a.name ~= b.name then return a.name < b.name end
				return tostring(a.id) < tostring(b.id)
			end)
			petsData = list
			petsLoaded = true
			if onDone then pcall(onDone) end
		end)
	end)
end

local function mySide()
	if not session then return nil end
	if session.RequesterUser == plr then return "Requester" end
	if session.ReceiverUser == plr then return "Receiver" end
	return nil
end

local function otherPlayer()
	if not session then return nil end
	if session.RequesterUser == plr then return session.ReceiverUser end
	if session.ReceiverUser == plr then return session.RequesterUser end
	return nil
end

local function ourItems()
	if not session then return {} end
	local side = mySide()
	if side == "Requester" then return session.RequesterItems or {} end
	if side == "Receiver" then return session.ReceiverItems or {} end
	return {}
end

local function ourFlag()
	if not session then return false end
	if mySide() == "Requester" then return session.RequesterConfirmed == true end
	if mySide() == "Receiver" then return session.ReceiverConfirmed == true end
	return false
end

local function otherFlag()
	if not session then return false end
	if mySide() == "Requester" then return session.ReceiverConfirmed == true end
	if mySide() == "Receiver" then return session.RequesterConfirmed == true end
	return false
end

local function shortId(id)
	return tostring(id):sub(1, 8)
end

local function updateStatus()
	if not Status or not Status.Parent then return end
	if not session then
		Status.Text = "сессия: нет. Выбери игрока → ЗАПРОСИТЬ ТРЕЙД."
		Status.TextColor3 = CLR.dim
		return
	end
	local other = otherPlayer()
	local items = ourItems()
	local added = 0
	for _, _ in pairs(items) do added = added + 1 end
	local sel = 0
	for _, on in pairs(selectedPets) do if on then sel = sel + 1 end end
	local flags = ""
	if ourFlag() then flags = " [мы: ПОДТВ." end
	if otherFlag() then
		if flags == "" then flags = " [партнёр: ПОДТВ." else flags = flags .. " · партнёр: ПОДТВ." end
	end
	if flags ~= "" then flags = flags .. "]" end
	Status.Text = ("сессия %s | %s | %s | в трейде=%d | выбрано=%d%s"):format(
		shortId(session.Id or "?"),
		tostring(session.Status or "?"),
		mySide() == "Requester" and "мы=requester" or (mySide() == "Receiver" and "мы=receiver" or "мы=? bit"),
		added,
		sel,
		flags
	)
	Status.TextColor3 = (session.Status == "Confirmed") and CLR.ok
		or (session.Status == "Accepted") and CLR.busy
		or (session.Status == "Cancelled") and CLR.bad
		or CLR.text
	if other then
		Status.Text = Status.Text .. "  ·  партнёр: " .. other.DisplayName
	end
end

local function refreshPets()
	if not petsLoaded then
		fetchPetsAsync(function() end)
		local t0 = os.clock()
		while not petsLoaded and os.clock() - t0 < 5 do task.wait(0.1) end
	end
	if petsLoaded then
		renderPets()
	end
end

local function flash(btn, color)
	if not btn or not btn.Parent then return end
	local old = btn.BackgroundColor3
	btn.BackgroundColor3 = color or CLR.ok
	task.delay(1.2, function()
		if btn and btn.Parent and not unloaded then
			btn.BackgroundColor3 = old
		end
	end)
end

local function requestTrade()
	if not TradeRequest then return end
	if not selectedPlayer or not selectedPlayer.Parent then
		flash(BtnRequest, CLR.bad)
		return
	end
	task.spawn(function()
		local ok = pcall(function() return TradeRequest:InvokeServer(selectedPlayer) end)
		flash(BtnRequest, ok and CLR.ok or CLR.bad)
		if not ok then
			pcall(print, "[OPTrade] Trade_Request упал (возможно, сессия уже есть)")
		end
	end)
end

local function acceptIncoming()
	if not TradeAccept then return end
	if not session or session.Status ~= "Sent" then
		flash(BtnAccept, CLR.bad)
		return
	end
	if session.RequesterUser == plr then
		flash(BtnAccept, CLR.bad)
		return
	end
	task.spawn(function()
		local ok = pcall(function() return TradeAccept:InvokeServer(session.RequesterUser) end)
		flash(BtnAccept, ok and CLR.ok or CLR.bad)
	end)
end

local function cancelSession()
	if not TradeCancel or not session then
		flash(BtnCancel, CLR.bad)
		return
	end
	if session.RequesterUser == plr and session.Status == "Sent" and TradeCancelRequest then
		task.spawn(function()
			pcall(function() return TradeCancelRequest:InvokeServer(session.ReceiverUser) end)
		end)
		return
	end
	task.spawn(function()
		local ok = pcall(function() return TradeCancel:InvokeServer({ Id = session.Id }) end)
		flash(BtnCancel, ok and CLR.ok or CLR.bad)
	end)
end

local function alreadyInTrade(petId)
	if not session then return false end
	local function scan(list)
		for _, v in pairs(list or {}) do
			if v.Id == petId then return true end
		end
		return false
	end
	return scan(session.RequesterItems) or scan(session.ReceiverItems)
end

local function addSelected(manual)
	if not TradeAddItem or not session then
		if manual then flash(BtnAdd, CLR.bad) end
		return 0
	end
	if session.Status == "Confirmed" or session.Status == "Cancelled" then
		if manual then flash(BtnAdd, CLR.bad) end
		return 0
	end
	local toAdd = {}
	for petId, on in pairs(selectedPets) do
		if on and not alreadyInTrade(petId) and not addedByUs[petId] then
			table.insert(toAdd, petId)
		end
	end
	if #toAdd == 0 then
		if manual then flash(BtnAdd, CLR.ok) end
		return 0
	end
	local count = 0
	for _, petId in ipairs(toAdd) do
		rateLimit()
		local ok = pcall(function()
			return TradeAddItem:InvokeServer({ Id = session.Id }, { Id = petId, Category = "Pets" })
		end)
		if ok then
			count = count + 1
			addedByUs[petId] = true
		end
		if ADD_GAP > 0 then task.wait(ADD_GAP) end
	end
	if manual then flash(BtnAdd, count > 0 and CLR.ok or CLR.bad) end
	pcall(print, ("[OPTrade] добавлено петов: %d"):format(count))
	updateStatus()
	return count
end

local function confirmTrade()
	if not TradeConfirm or not session then
		flash(BtnConfirm, CLR.bad)
		return
	end
	if session.Status == "Confirmed" or session.Status == "Cancelled" then
		flash(BtnConfirm, CLR.bad)
		return
	end
	local sid = session.Id
	task.spawn(function()
		local ok = pcall(function() return TradeConfirm:InvokeServer({ Id = sid }) end)
		if ok then
			confirmedFired = true
			flash(BtnConfirm, CLR.ok)
		else
			flash(BtnConfirm, CLR.bad)
		end
	end)
end

-- авто-режим
local lastSessionId = nil
task.spawn(function()
	while not unloaded do
		task.wait(0.5)
		if session and session.Id ~= lastSessionId then
			lastSessionId = session.Id
			addedByUs = {}
			confirmedFired = false
			updateStatus()
		end
		if session and cfg.autoAdd and session.Status == "Accepted" then
			local any = false
			for _, on in pairs(selectedPets) do
				if on then any = true break end
			end
			if any then
				local n = addSelected(false)
				if n > 0 then updateStatus() end
			end
		end
		if session and cfg.autoConfirm and session.Status == "Accepted" and not confirmedFired and not ourFlag() then
			local anyAdded = false
			for _, _ in pairs(ourItems()) do anyAdded = true break end
			local anySelected = false
			for _, on in pairs(selectedPets) do
				if on then anySelected = true break end
			end
			if anyAdded or not anySelected then
				confirmTrade()
			end
		end
	end
end)

		end

--====================================================================
-- UNLOAD
--====================================================================
local function unload()
	if unloaded then return end
	unloaded = true
	pcall(function()
		cfg.x = math.floor(Window.AbsolutePosition.X)
		cfg.y = math.floor(Window.AbsolutePosition.Y)
	end)
	saveConfig()
	for i = 1, #conns do pcall(function() conns[i]:Disconnect() end) end
	pcall(function() ScreenGui:Destroy() end)
	if getgenv then getgenv().__SM_OPTRADE = nil end
end

if getgenv then
	getgenv().__SM_OPTRADE = {
		Unload = unload,
		requestTrade = requestTrade,
		acceptIncoming = acceptIncoming,
		addSelected = addSelected,
		confirmTrade = confirmTrade,
		cancelSession = cancelSession,
		refreshPets = refreshPets,
		getSession = function() return session end,
	}
end
